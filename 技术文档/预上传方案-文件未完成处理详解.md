# 预上传方案 - 文件未完成上传时的处理详解

## 📋 问题场景

**场景：** 用户选择了一个 100MB 的文件，文件开始上传。当文件只上传了 50MB（50%）时，用户点击了提交按钮。

**问题：**
1. 表单提交时，文件只上传了一半，如何处理？
2. 如何保证数据一致性？
3. 文件上传完成后如何关联到表单？
4. 如果文件上传失败怎么办？

---

## 🎯 解决方案核心思路

### 核心原则

1. **表单数据先提交** - 不等待文件上传完成
2. **使用临时ID + 文件Hash** - 标识未完成的文件
3. **文件继续后台上传** - 上传完成后关联到表单
4. **服务器端记录状态** - 跟踪文件上传进度

```typescript
/**
 * 方案1：预上传方案（处理未完成文件）
 */
async function submitFormWithPreUploadedFiles(
  formData: Record<string, any>
): Promise<{
  success: boolean
  data?: any
  pendingFiles?: Array<{ tempId: string; fileHash: string; progress: number }>
}> {
  // 收集所有文件的状态
  const fileStatuses: FormUploadResult[] = []
  const pendingFiles: Array<{
    tempId: string
    fileHash: string
    progress: number
    fileName: string
    fileSize: number
  }> = []
  const completedFiles: Array<{ fileId: string }> = []

  for (const file of fileList.value) {
    const status = fileStatusMap.get(file.id)
    if (status) {
      if (status.status === 'success' && status.fileId) {
        // ✅ 文件已上传完成，使用文件ID
        completedFiles.push({ fileId: status.fileId })
        formData[`file_${file.id}`] = status.fileId
      } else {
        // ⚠️ 文件未上传完成，使用临时ID + Hash
        pendingFiles.push({
          tempId: status.tempId || file.id,
          fileHash: status.fileHash || file.fileHash || '',
          progress: status.progress || file.progress,
          fileName: file.name,
          fileSize: file.size
        })
        formData[`file_${file.id}`] = status.tempId  // 使用临时ID
      }
    }
  }

  // 提交表单数据
  try {
    const response = await fetch('/api/form/submit', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        ...formData,
        // 标记未完成的文件
        _pendingFiles: pendingFiles.map(f => ({
          tempId: f.tempId,
          fileHash: f.fileHash,
          fileName: f.fileName,
          fileSize: f.fileSize,
          progress: f.progress
        })),
        // 标记已完成的文件
        _completedFiles: completedFiles.map(f => f.fileId),
        // 标记异步上传模式
        _asyncUpload: pendingFiles.length > 0
      })
    })

    if (!response.ok) {
      throw new Error('表单提交失败')
    }

    const result = await response.json()
    const formId = result.formId

    // 🔑 关键：文件继续在后台上传，上传完成后关联到表单
    if (pendingFiles.length > 0 && formId) {
      // 监听文件上传完成事件
      setupFileUploadCompletionListener(formId, pendingFiles)
    }

    return {
      success: true,
      data: result,
      pendingFiles: pendingFiles.length > 0 ? pendingFiles : undefined
    }
  } catch (error) {
    console.error('表单提交失败:', error)
    return {
      success: false,
      pendingFiles: pendingFiles.length > 0 ? pendingFiles : undefined
    }
  }
}
```

#### 1.2 监听文件上传完成

```typescript
/**
 * 设置文件上传完成监听器
 * 当文件上传完成时，自动关联到表单
 */
function setupFileUploadCompletionListener(
  formId: string,
  pendingFiles: Array<{ tempId: string; fileHash: string }>
): void {
  // 为每个未完成的文件设置监听器
  pendingFiles.forEach(({ tempId, fileHash }) => {
    const file = fileList.value.find(f => f.id === tempId)
    if (!file) return

    const task = uploadTasks.get(file.id)
    if (!task) return

    // 方式1：轮询检查文件状态
    const checkInterval = setInterval(async () => {
      const status = fileStatusMap.get(file.id)
      
      if (status?.status === 'success' && status.fileId) {
        // 文件上传完成，关联到表单
        clearInterval(checkInterval)
        await associateFileToForm(formId, tempId, status.fileId)
      } else if (status?.status === 'error') {
        // 文件上传失败，通知用户
        clearInterval(checkInterval)
        console.error(`文件 ${file.name} 上传失败`)
        // 可以调用接口通知服务器文件上传失败
        await notifyFileUploadFailed(formId, tempId, fileHash)
      }
    }, 2000)  // 每2秒检查一次

    // 方式2：使用 WebSocket 监听（更高效）
    if (wsUrl) {
      const wsManager = getWebSocketManager(wsUrl)
      wsManager.onMessage((message) => {
        if (
          message.type === 'mergeComplete' &&
          message.fileId === file.id
        ) {
          // 文件合并完成，获取文件ID并关联
          getFileIdAndAssociate(formId, tempId, fileHash)
        } else if (
          message.type === 'error' &&
          message.fileId === file.id
        ) {
          // 文件上传失败
          notifyFileUploadFailed(formId, tempId, fileHash)
        }
      })
    }
  })
}

/**
 * 获取文件ID并关联到表单
 */
async function getFileIdAndAssociate(
  formId: string,
  tempId: string,
  fileHash: string
): Promise<void> {
  try {
    // 1. 获取文件ID
    const fileIdResponse = await fetch(`/api/upload/get-file-id?fileHash=${fileHash}`)
    if (!fileIdResponse.ok) {
      throw new Error('获取文件ID失败')
    }
    const { fileId } = await fileIdResponse.json()

    // 2. 关联文件到表单
    await associateFileToForm(formId, tempId, fileId)
    
    console.log(`文件 ${tempId} 已关联到表单 ${formId}`)
  } catch (error) {
    console.error('关联文件失败:', error)
    // 重试机制
    setTimeout(() => {
      getFileIdAndAssociate(formId, tempId, fileHash)
    }, 5000)
  }
}

/**
 * 关联文件到表单
 */
async function associateFileToForm(
  formId: string,
  tempId: string,
  fileId: string
): Promise<void> {
  try {
    const response = await fetch('/api/form/associate-file', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        formId,
        tempId,
        fileId
      })
    })

    if (!response.ok) {
      throw new Error('关联文件失败')
    }

    console.log(`文件 ${fileId} 已成功关联到表单 ${formId}`)
  } catch (error) {
    console.error('关联文件失败:', error)
    throw error
  }
}

/**
 * 通知服务器文件上传失败
 */
async function notifyFileUploadFailed(
  formId: string,
  tempId: string,
  fileHash: string
): Promise<void> {
  try {
    await fetch('/api/form/file-upload-failed', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        formId,
        tempId,
        fileHash
      })
    })
  } catch (error) {
    console.error('通知上传失败失败:', error)
  }
}
```

```javascript
// POST /api/form/submit
router.post('/submit', async (req, res) => {
  const { 
    title, 
    description, 
    _pendingFiles = [],      // 未完成的文件
    _completedFiles = [],    // 已完成的文件
    _asyncUpload            // 是否异步上传模式
  } = req.body
  
  try {
    // 创建表单记录
    const formId = generateFormId()
    const form = {
      id: formId,
      title,
      description,
      fileIds: _completedFiles,  // 已完成的文件ID
      pendingFiles: _pendingFiles.map(f => ({
        tempId: f.tempId,
        fileHash: f.fileHash,
        fileName: f.fileName,
        fileSize: f.fileSize,
        progress: f.progress,
        status: 'uploading'  // 状态：uploading | completed | failed
      })),
      status: _asyncUpload ? 'pending' : 'completed',  // pending: 等待文件上传完成
      createdAt: Date.now(),
      updatedAt: Date.now()
    }
    
    // 保存到数据库
    await db.forms.insert(form)
    
    // 如果有关联的文件上传任务，记录关联关系
    if (_asyncUpload && _pendingFiles.length > 0) {
      for (const pendingFile of _pendingFiles) {
        // 在文件上传任务表中记录表单关联
        await db.fileUploadTasks.insert({
          fileHash: pendingFile.fileHash,
          tempId: pendingFile.tempId,
          formId: formId,
          status: 'uploading',
          progress: pendingFile.progress,
          createdAt: Date.now()
        })
      }
    }
    
    res.json({
      success: true,
      formId,
      message: '表单提交成功',
      pendingFilesCount: _pendingFiles.length
    })
  } catch (error) {
    console.error('[FORM SUBMIT ERROR] - ', error)
    res.status(500).json({
      success: false,
      message: '表单提交失败：' + error.message
    })
  }
})
```

#### 2.2 文件关联接口

```javascript
// POST /api/form/associate-file
router.post('/associate-file', async (req, res) => {
  const { formId, tempId, fileId } = req.body
  
  try {
    // 1. 查找表单
    const form = await db.forms.findOne({ id: formId })
    if (!form) {
      return res.status(404).json({ error: '表单不存在' })
    }
    
    // 2. 查找文件上传任务
    const uploadTask = await db.fileUploadTasks.findOne({
      formId,
      tempId
    })
    
    if (uploadTask) {
      // 更新任务状态
      await db.fileUploadTasks.update(
        { formId, tempId },
        {
          $set: {
            fileId,
            status: 'completed',
            completedAt: Date.now()
          }
        }
      )
    }
    
    // 3. 更新表单中的文件ID
    const fileIds = form.fileIds || []
    fileIds.push(fileId)
    
    // 4. 从待处理列表中移除
    const updatedPendingFiles = form.pendingFiles.filter(
      f => f.tempId !== tempId
    )
    
    // 5. 更新表单状态
    const isAllCompleted = updatedPendingFiles.length === 0
    
    await db.forms.update(
      { id: formId },
      {
        $set: {
          fileIds: fileIds,
          pendingFiles: updatedPendingFiles,
          status: isAllCompleted ? 'completed' : 'pending',
          updatedAt: Date.now()
        }
      }
    )
    
    // 6. 如果所有文件都上传完成，触发完成事件（可选）
    if (isAllCompleted) {
      // 可以发送通知、触发后续流程等
      await notifyFormCompleted(formId)
    }
    
    res.json({
      success: true,
      message: '文件关联成功',
      formStatus: isAllCompleted ? 'completed' : 'pending'
    })
  } catch (error) {
    console.error('[ASSOCIATE FILE ERROR] - ', error)
    res.status(500).json({
      success: false,
      message: '文件关联失败：' + error.message
    })
  }
})
```

#### 2.3 文件上传失败处理接口

```javascript
// POST /api/form/file-upload-failed
router.post('/file-upload-failed', async (req, res) => {
  const { formId, tempId, fileHash } = req.body
  
  try {
    // 1. 更新文件上传任务状态
    await db.fileUploadTasks.update(
      { formId, tempId },
      {
        $set: {
          status: 'failed',
          failedAt: Date.now()
        }
      }
    )
    
    // 2. 更新表单中的文件状态
    const form = await db.forms.findOne({ id: formId })
    if (form) {
      const updatedPendingFiles = form.pendingFiles.map(f => {
        if (f.tempId === tempId) {
          return {
            ...f,
            status: 'failed'
          }
        }
        return f
      })
      
      await db.forms.update(
        { id: formId },
        {
          $set: {
            pendingFiles: updatedPendingFiles,
            updatedAt: Date.now()
          }
        }
      )
    }
    
    // 3. 可以发送通知给用户（可选）
    await notifyUserFileUploadFailed(formId, tempId)
    
    res.json({
      success: true,
      message: '文件上传失败已记录'
    })
  } catch (error) {
    console.error('[FILE UPLOAD FAILED ERROR] - ', error)
    res.status(500).json({
      success: false,
      message: '记录失败：' + error.message
    })
  }
})
```
